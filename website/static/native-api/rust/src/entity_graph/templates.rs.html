6/// TypeScript entity type interface template.
7///
8/// Context variables (set by `generate`):
9/// - `entity_name` : PascalCase entity name          (string)
10/// - `fields`      : array of FieldCtx               (see below)
11/// - `relations`   : array of RelationCtx
12pub const ENTITY_TYPES_TS: &str = r#"// AUTO-GENERATED by entity-graph-cli {{ cli_version }}
13// Source: {{ schema_path }}
14// Do not edit by hand — regenerate with `entity-graph generate`.
15
16{% for entity in entities %}
17// ── {{ entity.pascal_name }} ──────────────────────────────────────────────────
18
19export interface {{ entity.pascal_name }} {
20{% for field in entity.fields %}  {{ field.name }}{% if field.optional %}?{% endif %}: {{ field.ts_type }};
21{% endfor %}
22}
23
24export type Create{{ entity.pascal_name }}Input = Omit<{{ entity.pascal_name }}, '{{ entity.primary_key }}'{% if entity.has_auto_fields %} | {{ entity.auto_field_names }}{% endif %}>;
25export type Update{{ entity.pascal_name }}Input = Partial<Create{{ entity.pascal_name }}Input>;
26
27{% endfor %}
28"#;

30/// TypeScript transport registration stub template.
31///
32/// Emits a `registerTransports.ts` that calls `registerEntityTransport` for
33/// every entity in the IR, ready for the user to fill in supabase/REST details.
34pub const REGISTER_TRANSPORTS_TS: &str = r#"// AUTO-GENERATED by entity-graph-cli {{ cli_version }}
35// Source: {{ schema_path }}
36// Do not edit by hand — regenerate with `entity-graph generate`.
37
38/**
39 * Call this once at application boot (before any component mounts) to register
40 * all entity transports with the @prometheus-ags/entity-graph-core store.
41 *
42 * Replace the `makeRestTransport` calls with the transport appropriate for
43 * your backend (GraphQL, Supabase, etc.).
44 */
45
46import {
47  registerEntityTransport,
48  makeRestTransport,
49} from '@prometheus-ags/entity-graph-core';
50{% for entity in entities %}
51import type { {{ entity.pascal_name }} } from './entity-types';
52{% endfor %}
53
54export function registerTransports(baseUrl: string): void {
55{% for entity in entities %}
56  registerEntityTransport<{{ entity.pascal_name }}>('{{ entity.pascal_name }}', makeRestTransport({
57    endpoint: `${baseUrl}/{{ entity.table }}`,
58    primaryKey: '{{ entity.primary_key }}',
59  }));
60{% endfor %}
61}
62"#;
