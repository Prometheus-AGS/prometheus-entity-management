19/// Code-generation targets.
20#[derive(Debug, Clone, clap::ValueEnum)]
21pub enum Target {
22    /// Emit TypeScript types + a `registerEntityTransport` stub.
23    React,
24}

26/// Arguments for the `generate` subcommand (parsed by `clap` in main).
27#[derive(Debug, clap::Args)]
28pub struct GenerateArgs {
29    /// Code-generation target.
30    #[arg(short, long, value_enum, default_value = "react")]
31    pub target: Target,
32
33    /// Path to the SDL JSON schema (produced by `init` or `parseSdlJson`).
34    #[arg(short, long, default_value = "schema.json")]
35    pub schema: PathBuf,
36
37    /// Output directory for generated files.
38    #[arg(short, long, default_value = "generated")]
39    pub out: PathBuf,
40}

42// ── Template context types ─────────────────────────────────────────────────────

44/// Per-field data passed into Tera templates.
45#[derive(Debug, Serialize)]
46struct FieldCtx {
47    name: String,
48    ts_type: String,
49    optional: bool,
50    primary: bool,
51    auto: bool,
52}
53
54/// Per-relation data passed into Tera templates.
55#[derive(Debug, Serialize)]
56struct RelationCtx {
57    name: String,
58    kind: String,
59    target_pascal: String,
60    foreign_key: Option<String>,
61}
62
63/// Per-entity data passed into Tera templates.
64#[derive(Debug, Serialize)]
65struct EntityCtx {
66    pascal_name: String,
67    snake_name: String,
68    table: String,
69    primary_key: String,
70    fields: Vec<FieldCtx>,
71    relations: Vec<RelationCtx>,
72    has_auto_fields: bool,
73    /// Quoted comma-separated names of auto fields for Omit<>.
74    auto_field_names: String,
75}
76
77// ── Helpers ───────────────────────────────────────────────────────────────────
78
79fn build_entity_ctx(entity: &IrEntity) -> EntityCtx {
80    let fields: Vec<FieldCtx> = entity
81        .fields
82        .iter()
83        .map(|f| FieldCtx {
84            name: f.name.clone(),
85            ts_type: f.ts_type_string(),
86            optional: f.is_optional(),
87            primary: f.primary,
88            auto: f.auto,
89        })
90        .collect();
91
92    let auto_fields: Vec<&str> = entity
93        .fields
94        .iter()
95        .filter(|f| f.auto)
96        .map(|f| f.name.as_str())
97        .collect();
98    let has_auto_fields = !auto_fields.is_empty();
99    let auto_field_names = auto_fields
100        .iter()
101        .map(|n| format!("'{}'", n))
102        .collect::<Vec<_>>()
103        .join(" | ");
104
105    let relations: Vec<RelationCtx> = entity
106        .relations
107        .iter()
108        .map(|r| RelationCtx {
109            name: r.name.clone(),
110            kind: match r.kind {
111                RelationKind::BelongsTo => "belongsTo",
112                RelationKind::HasMany => "hasMany",
113                RelationKind::ManyToMany => "manyToMany",
114            }
115            .to_owned(),
116            target_pascal: r.target.to_case(Case::Pascal),
117            foreign_key: r.foreign_key.clone(),
118        })
119        .collect();
120
121    EntityCtx {
122        pascal_name: entity.name.to_case(Case::Pascal),
123        snake_name: entity.name.to_case(Case::Snake),
124        table: entity.table.clone(),
125        primary_key: entity.primary_key.clone(),
126        fields,
127        relations,
128        has_auto_fields,
129        auto_field_names,
130    }
131}

133/// Build the shared Tera context for all templates.
134fn build_tera_ctx(ir: &EntityGraphIR, schema_path: &str) -> TeraCtx {
135    let entities: Vec<EntityCtx> = ir.entities.iter().map(build_entity_ctx).collect();
136    let mut ctx = TeraCtx::new();
137    ctx.insert("entities", &entities);
138    ctx.insert("schema_path", schema_path);
139    ctx.insert("cli_version", env!("CARGO_PKG_VERSION"));
140    ctx
141}

143/// Render a Tera template string and return the output.
144fn render(template_name: &str, template_src: &str, ctx: &TeraCtx) -> Result<String> {
145    let mut tera = Tera::default();
146    tera.add_raw_template(template_name, template_src)
147        .with_context(|| format!("failed to parse template '{template_name}'"))?;
148    tera.render(template_name, ctx)
149        .with_context(|| format!("failed to render template '{template_name}'"))
150}

152// ── Entry point ───────────────────────────────────────────────────────────────
153
154/// Execute the `generate` subcommand.
155pub fn run(args: &GenerateArgs) -> Result<()> {
156    // 1. Load + parse the schema.
157    // Accept both the human-editable SDL source format (written by `init`) and
158    // the EntityGraphIR array format produced by `parseSdlJson()` in the TS SDL
159    // package. We try the IR format first; on failure fall back to SDL source.
160    let schema_str = std::fs::read_to_string(&args.schema)
161        .with_context(|| format!("cannot read schema from {}", args.schema.display()))?;
162
163    let ir: EntityGraphIR = {
164        // Attempt IR format first (entities is an array).
165        let ir_attempt: Result<EntityGraphIR> = serde_json::from_str(&schema_str)
166            .with_context(|| format!("IR parse attempt for {}", args.schema.display()));
167        match ir_attempt {
168            Ok(ir) if !ir.entities.is_empty() || ir.version == "1.0" => ir,
169            _ => {
170                // Fall back to SDL source format (entities is an object keyed by name).
171                sdl_parser::parse_sdl_source(&schema_str)
172                    .with_context(|| format!("invalid SDL JSON in {}", args.schema.display()))?
173            }
174        }
175    };
176
177    let schema_path = args.schema.to_string_lossy().into_owned();
178
179    // 2. Prepare output directory.
180    if !args.out.exists() {
181        std::fs::create_dir_all(&args.out)
182            .with_context(|| format!("cannot create output dir {}", args.out.display()))?;
183    }
184
185    let ctx = build_tera_ctx(&ir, &schema_path);
186
187    match args.target {
188        Target::React => generate_react(&args.out, &ctx)?,
189    }
190
191    Ok(())
192}
193
194fn generate_react(out: &std::path::Path, ctx: &TeraCtx) -> Result<()> {
195    // entity-types.ts
196    let types_src = render("entity-types.ts.tera", templates::ENTITY_TYPES_TS, ctx)?;
197    let types_path = out.join("entity-types.ts");
198    std::fs::write(&types_path, &types_src)
199        .with_context(|| format!("cannot write {}", types_path.display()))?;
200    println!("Wrote {}", types_path.display());
201
202    // register-transports.ts
203    let reg_src = render(
204        "register-transports.ts.tera",
205        templates::REGISTER_TRANSPORTS_TS,
206        ctx,
207    )?;
208    let reg_path = out.join("register-transports.ts");
209    std::fs::write(&reg_path, &reg_src)
210        .with_context(|| format!("cannot write {}", reg_path.display()))?;
211    println!("Wrote {}", reg_path.display());
212
213    // index.ts
214    let index_src = render("index.ts.tera", templates::INDEX_TS, ctx)?;
215    let index_path = out.join("index.ts");
216    std::fs::write(&index_path, &index_src)
217        .with_context(|| format!("cannot write {}", index_path.display()))?;
218    println!("Wrote {}", index_path.display());
219
220    println!();
221    println!("Done. {} entity/entities generated.", {
222        // entity count from context — re-parse to avoid extra dep
223        let dummy: serde_json::Value =
224            serde_json::to_value(ctx.get("entities").unwrap_or(&tera::Value::Null))
225                .unwrap_or_default();
226        dummy.as_array().map(|a| a.len()).unwrap_or(0)
227    });
228
229    Ok(())
230}
